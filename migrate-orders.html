return fallback; } } function writeStorage(key, value) { try { localStorage.setItem(key, JSON.stringify(value)); return true; } catch (error) { console.warn('Storage write failed:', error); return false; } } function makeId(prefix = 'gt') { if (window.crypto && crypto.randomUUID) { return `${prefix}_${crypto.randomUUID()}`; } return `${prefix}_${Date.now().toString(36)}_${Math.random().toString(36).slice(2, 10)}`; const emailExists = profiles.some((profile) => String(profile.email || '').toLowerCase() === cleanEmail); const usernameExists = profiles.some((profile) => String(profile.username || '').toLowerCase() === cleanName.toLowerCase()); profiles.push(user); writeStorage(STORAGE_PROFILES, profiles); return { success: true, userId: user.id, requiresEmailConfirmation: false, user: { id: user.id, email: user.email, user_metadata: { username: user.username } } }; } async function loginUser(email, password) { const cleanEmail = String(email || '').trim().toLowerCase(); if (!cleanEmail || !password) { return { success: false, error: '邮箱和密码不能为空' }; } const profiles = ensureArray(readStorage(STORAGE_PROFILES, [])); const profile = profiles.find((item) => String(item.email || '').toLowerCase() === cleanEmail); if (!profile) { return { success: false, error: '账户不存在，请先注册' }; } const session = { user: { id: profile.id, email: profile.email, user_metadata: { username: profile.username } } }; writeStorage(STORAGE_SESSION, session); return { success: true, user: session.user }; } async function logoutUser() { localStorage.removeItem(STORAGE_SESSION); return { success: true }; } async function getCurrentUser() { const session = readStorage(STORAGE_SESSION, null); if (!session || !session.user) return null; return session.user; } async function createUserProfile(userId, username, email) { const profiles = ensureArray(readStorage(STORAGE_PROFILES, [])); const existing = profiles.find((profile) => profile.id === userId); return { success: true, order: normalized }; } async function getUserOrders(userId) { const orders = ensureArray(readStorage(STORAGE_ORDERS, [])).filter((order) => String(order.user_id) === String(userId));

return true; } catch (error) { console.warn('Unable to write storage:', error); return false; } } function ensureArray(value) { return Array.isArray(value) ? value : []; } function makeId(prefix = 'gt') { if (window.crypto && crypto.randomUUID) { return `${prefix}_${crypto.randomUUID()}`; } if (String(password).length < 6) { return { success: false, error: '密码至少需要 6 个字符' }; } const profiles = ensureArray(readStorage(STORAGE_PROFILES, [])); if (profiles.some((profile) => normalizeEmail(profile.email) === cleanEmail)) { return { success: false, error: '该邮箱已注册' }; } if (profiles.some((profile) => String(profile.username || '').toLowerCase() === cleanUsername.toLowerCase())) { return { success: false, error: '用户名已存在' }; } const profile = { id: makeId('user'), username: cleanUsername, email: cleanEmail, createdAt: new Date().toISOString() }; profiles.push(profile); writeStorage(STORAGE_PROFILES, profiles); const session = { user: { id: profile.id, email: profile.email, user_metadata: { username: profile.username } } }; writeStorage(STORAGE_SESSION, session); return { success: true, userId: profile.id, requiresEmailConfirmation: false, user: session.user }; } async function loginUser(email, password) { const cleanEmail = normalizeEmail(email); if (!cleanEmail || !password) { return { success: false, error: '邮箱和密码不能为空' }; } const profiles = ensureArray(readStorage(STORAGE_PROFILES, [])); const profile = profiles.find((item) => normalizeEmail(item.email) === cleanEmail); if (!profile) { return { success: false, error: '账户不存在，请先注册' }; } const session = { user: { id: profile.id, email: profile.email, user_metadata: { username: profile.username } } }; writeStorage(STORAGE_SESSION, session); return { success: true, user: session.user }; } async function logoutUser() { localStorage.removeItem(STORAGE_SESSION); return { success: true }; } async function getCurrentUser() { const session = readStorage(STORAGE_SESSION, null); return session && session.user ? session.user : null; return { success: true, order: normalized }; } async function getUserOrders(userId) { const orders = ensureArray(readStorage(STORAGE_ORDERS, [])).filter((order) => String(order.user_id) === String(userId)); if (!cleanName) return null; const profiles = ensureArray(readStorage(STORAGE_PROFILES, [])); return profiles.find((profile) => String(profile.username || '').toLowerCase() === cleanName.toLowerCase()) || null;
